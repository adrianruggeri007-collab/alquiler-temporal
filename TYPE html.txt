<script>
    const WHATSAPP_OWNER = "5493624708675";
    const SUPABASE_URL = "https://ougnzitbeervypchwxmx.supabase.co";
    const SUPABASE_ANON_KEY = "sb_publishable_oiEh5_vvS1fPJr-TBLil1A_q-VFWmFD";
    const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    let currentDate = new Date();
    let selectedDates = [];
    let bookings = [];

    const monthNames = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
    const dayNames = ["Dom", "Lun", "Mar", "Mié", "Jue", "Vie", "Sáb"];

    function getTodayStr() {
        const now = new Date();
        return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(now.getDate()).padStart(2, '0')}`;
    }

    function isPastDate(dateStr) {
        return dateStr < getTodayStr();
    }

    async function loadBookings() {
        const { data, error } = await supabaseClient
            .from('reservas')
            .select('id, nombre, telefono, huespedes, fechas, estado')
            .order('id', { ascending: false });

        if (error) {
            console.error("Error al cargar reservas:", error);
            return;
        }

        bookings = (data || []).map(row => ({
            id: Number(row.id),
            name: row.nombre,
            phone: row.telefono,
            guests: row.huespedes,
            dates: row.fechas || [],
            status: row.estado || 'pendiente'
        }));

        renderCalendar();
    }

    function renderCalendar() {
        const year = currentDate.getFullYear();
        const month = currentDate.getMonth();
        document.getElementById('monthYear').innerText = `${monthNames[month]} ${year}`;

        const firstDay = new Date(year, month, 1).getDay();
        const daysInMonth = new Date(year, month + 1, 0).getDate();
        const calendarGrid = document.getElementById('calendarDays');
        calendarGrid.innerHTML = '';

        dayNames.forEach(d => {
            calendarGrid.innerHTML += `<div class="day-name">${d}</div>`;
        });

        for (let i = 0; i < firstDay; i++) {
            calendarGrid.innerHTML += `<div class="day empty"></div>`;
        }

        const bookedDates = bookings.flatMap(b => b.dates);

        for (let day = 1; day <= daysInMonth; day++) {
            const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
            let classes = 'day';

            if (bookedDates.includes(dateStr)) {
                classes += ' booked';
            } else if (isPastDate(dateStr)) {
                classes += ' past';
            } else if (selectedDates.includes(dateStr)) {
                classes += ' selected';
            }

            calendarGrid.innerHTML += `<div class="${classes}" onclick="toggleDate('${dateStr}')">${day}</div>`;
        }
        updateAdminList();
    }

    function toggleDate(dateStr) {
        if (isPastDate(dateStr)) return;

        const bookedDates = bookings.flatMap(b => b.dates);
        if (bookedDates.includes(dateStr)) return;

        if (selectedDates.includes(dateStr)) {
            selectedDates = selectedDates.filter(d => d !== dateStr);
        } else {
            selectedDates.push(dateStr);
        }
        selectedDates.sort();
        document.getElementById('selectedDatesText').value = selectedDates.join(', ');
        renderCalendar();
    }

    function changeMonth(step) {
        currentDate.setMonth(currentDate.getMonth() + step);
        renderCalendar();
    }

    async function saveBooking(e) {
        e.preventDefault();
        selectedDates = selectedDates.filter(d => !isPastDate(d));
        document.getElementById('selectedDatesText').value = selectedDates.join(', ');

        if (selectedDates.length === 0) {
            alert("Por favor seleccioná al menos un día disponible desde hoy en adelante.");
            return;
        }

        const submitBtn = document.getElementById('btnSubmit');
        submitBtn.disabled = true;

        const name = document.getElementById('guestName').value;
        const phone = document.getElementById('guestPhone').value;
        const guests = parseInt(document.getElementById('guestCount').value, 10) || 2;
        const datesToSave = [...selectedDates];

        const { error } = await supabaseClient
            .from('reservas')
            .insert([{
                nombre: name,
                telefono: phone,
                huespedes: guests,
                fechas: datesToSave
            }]);

        submitBtn.disabled = false;

        if (error) {
            console.error("Error al guardar reserva:", error);
            alert("No se pudo registrar la reserva. Error: " + error.message);
            return;
        }

        const mensaje = `¡Nueva solicitud de reserva!\n\n👤 Nombre: ${name}\n📞 Teléfono: ${phone}\n👥 Huéspedes: ${guests}\n📅 Fechas: ${datesToSave.join(', ')}`;
        window.open(`https://wa.me/${WHATSAPP_OWNER}?text=${encodeURIComponent(mensaje)}`, '_blank');

        alert(`¡Reserva registrada con éxito para ${name}!`);
        selectedDates = [];
        document.getElementById('bookingForm').reset();
        await loadBookings();
    }

    function toggleAdmin() {
        const box = document.getElementById('adminBox');
        if (box.style.display === 'block') {
            box.style.display = 'none';
            return;
        }
        const pass = prompt("Ingresá la contraseña de propietario (por defecto: 1234):");
        if (pass === "1234") {
            box.style.display = 'block';
        } else {
            alert("Contraseña incorrecta.");
        }
    }

    async function deleteBooking(id) {
        if (confirm("¿Seguro que querés eliminar esta reserva y liberar las fechas?")) {
            const numericId = Number(id);
            const { error } = await supabaseClient
                .from('reservas')
                .delete()
                .eq('id', numericId);

            if (error) {
                console.error("Error al eliminar reserva:", error);
                alert("No se pudo eliminar la reserva: " + error.message);
                return;
            }

            alert("Reserva eliminada con éxito.");
            await loadBookings();
        }
    }

    function updateAdminList() {
        const list = document.getElementById('bookingsList');
        list.innerHTML = '';
        if (bookings.length === 0) {
            list.innerHTML = '<li>No hay reservas cargadas todavía.</li>';
            return;
        }
        bookings.forEach(booking => {
            const li = document.createElement('li');
            li.innerHTML = `<span><strong>${booking.name}</strong> - ${booking.dates.join(', ')}</span> <button class="btn-delete" onclick="deleteBooking(${booking.id})">Eliminar</button>`;
            list.appendChild(li);
        });
    }

    loadBookings();
